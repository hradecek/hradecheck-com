{{- $src := .Get "src" -}} {{- $alt := .Get "alt" | default "" -}} {{- $caption := .Get "caption" -}} {{- $light := resources.Get $src -}} {{- if not $light -}} {{- errorf "image shortcode: asset %q not found (in %s)" $src $.Position -}} {{- end -}} {{/* Derive the dark variant: insert _dark before the language/extension suffix, e.g. img/flow.svg -> img/flow_dark.svg, img/flow.sk.svg -> img/flow_dark.sk.svg. The directory contains no dots, so splitting on "." isolates the stem safely. */}} {{- $stem := index (split $src ".") 0 -}} {{- $rest := strings.TrimPrefix $stem $src -}} {{- $dark := resources.Get (printf "%s_dark%s" $stem $rest) -}}
